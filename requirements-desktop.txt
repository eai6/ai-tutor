# Offline desktop build (Linux / Windows / macOS).
#
#   pip install -r requirements-core.txt -r requirements-desktop.txt
#
# Deliberately NOT merged into requirements.txt: the server has no use for a
# webview or an installer builder, and requirements.txt is what the Azure
# image installs.
#
# What this build DOES NOT install, and why that is the point:
#
#   torch                 372 MB   \  required only by sentence-transformers,
#   transformers           51 MB    >  which is required by nothing else. The
#   sentence-transformers 3.8 MB   /   desktop build embeds via onnxruntime.
#
# onnxruntime and tokenizers are already in requirements-core.txt (chromadb /
# faster-whisper pull them anyway), so the only thing the ONNX path adds is the
# 87 MB exported model in models/minilm-l6-v2/ — a build artifact, not a wheel.
# Net: roughly 340 MB smaller than the sentence-transformers path.
#
# Build the encoder before packaging:
#     python scripts/export_minilm_onnx.py     (needs torch + onnx; build box only)
#
# Plan: memory/desktop_offline_app_plan.md

# Native window over the local Django app. Uses the platform webview:
# WKWebView (macOS), WebView2 (Windows), WebKitGTK (Linux).
pywebview==6.2.1

# WSGI server for the loopback app. Pure Python and works on Windows, which
# gunicorn does not.
waitress==3.0.2

# Freezes the app into a distributable binary.
pyinstaller==6.21.0

# Linux only: pywebview needs GTK bindings, which are system packages rather
# than wheels. On Debian/Ubuntu:
#   sudo apt install python3-gi python3-gi-cairo gir1.2-gtk-4.0 \
#                    gir1.2-webkit-6.0
# Windows needs the WebView2 runtime, which ships with Windows 11 and is a
# redistributable on Windows 10 — the installer must check for it.
